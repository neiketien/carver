<?php
require_once 'includes/config.php';

$pdo = getDBConnection();

// Get URL parameters
$categorySlug = $_GET['category'] ?? '';
$tagSlug = $_GET['tag'] ?? '';
$articleSlug = $_GET['slug'] ?? '';

// Fetch article
$stmt = $pdo->prepare("
    SELECT a.*, 
           c.name as category_name, c.slug as category_slug,
           u.full_name as author_name, u.avatar as author_avatar
    FROM articles a
    JOIN categories c ON a.category_id = c.id
    JOIN users u ON a.user_id = u.id
    WHERE a.slug = ? AND a.status = 'published'
");
$stmt->execute([$articleSlug]);
$article = $stmt->fetch();

if (!$article) {
    header("HTTP/1.0 404 Not Found");
    include '404.php';
    exit;
}

$readingTime = calculateReadingTime($article['content']);

// Increment view count
$updateViewStmt = $pdo->prepare("UPDATE articles SET view_count = view_count + 1 WHERE id = ?");
$updateViewStmt->execute([$article['id']]);

// Get article tags
$tagsStmt = $pdo->prepare("
    SELECT t.name, t.slug 
    FROM tags t
    JOIN article_tags at ON t.id = at.tag_id
    WHERE at.article_id = ?
");
$tagsStmt->execute([$article['id']]);
$tags = $tagsStmt->fetchAll();

// Get related articles (same category, different article)
$relatedStmt = $pdo->prepare("
    SELECT a.id, a.title, a.slug, a.published_at,
           c.slug as category_slug,
           (SELECT t.slug FROM article_tags at JOIN tags t ON at.tag_id = t.id WHERE at.article_id = a.id LIMIT 1) as first_tag
    FROM articles a
    JOIN categories c ON a.category_id = c.id
    WHERE a.category_id = ? AND a.id != ? AND a.status = 'published'
    ORDER BY a.published_at DESC
    LIMIT 4
");
$relatedStmt->execute([$article['category_id'], $article['id']]);
$relatedArticles = $relatedStmt->fetchAll();

// Format article URL
$articleUrl = SITE_URL . "/{$article['category_slug']}/{$tagSlug}/{$article['slug']}";
$featuredImageUrl = UPLOAD_URL . $article['featured_image'];
$twitterCardImage = !empty($article['twitter_card_image']) 
    ? UPLOAD_URL . $article['twitter_card_image'] 
    : $featuredImageUrl;

// Calculate time indicators
$publishedAgo = timeAgo($article['published_at']);
$updatedAgo = ($article['updated_at'] !== $article['created_at']) ? timeAgo($article['updated_at']) : null;

$metaDescription = $article['meta_description'];

if (empty($metaDescription)) {
    // Check if subheading is longer than 60 chars
    if (mb_strlen($article['subheading']) > 60) {
        // Cut at 60 chars and add "..."
        $metaDescription = mb_substr($article['subheading'], 0, 60) . '...';
    } else {
        $metaDescription = $article['subheading'];
    }
}
