<?php
require_once 'includes/config.php';
requireAuth();

$pdo = getDBConnection();
$user = getCurrentUser();
$message = '';
$error = '';

$pendingCount = 0;
// Only query if the user has a role that cares about pending articles (Editor/Admin)
if (hasRole('editor') || hasRole('admin')) {
    $pendingCount = $pdo->query("SELECT COUNT(*) FROM articles WHERE status = 'pending'")->fetchColumn();
}

// Check if editing an existing article
$articleId = $_GET['id'] ?? null;
$article = null;
$articleTags = '';

if ($articleId) {
    $stmt = $pdo->prepare("SELECT * FROM articles WHERE id = ?");
    $stmt->execute([$articleId]);
    $article = $stmt->fetch();

    // Permissions check: Writers can only edit their own
    if ($user['role'] === 'writer' && $article['user_id'] !== $user['id']) {
        header('Location: manage-articles.php');
        exit;
    }

    // Fetch existing tags
    $tagStmt = $pdo->prepare("SELECT t.name FROM tags t JOIN article_tags at ON t.id = at.tag_id WHERE at.article_id = ?");
    $tagStmt->execute([$articleId]);
    $articleTags = implode(', ', $tagStmt->fetchAll(PDO::FETCH_COLUMN));
}

$tagsInput = $_POST['article-tags'] ?? '';
$keywordsInput = $_POST['article-keywords'] ?? '';

// Handle Form Submission
if ($_SERVER['REQUEST_METHOD'] === 'POST') {
    $title = trim($_POST['article-title']);
    $subheading = trim($_POST['article-subheading']);
    $categoryId = $_POST['article-category']; // In DB this is int, form sends string value (slug). Need to resolve.
    $content = $_POST['article-content'];
    $tagsInput = $_POST['article-tags'];
    $caption = $_POST['article-photo-caption'];
    $isTopStory = isset($_POST['top-story']) ? 1 : 0;
    $status = $_POST['status'] ?? 'draft'; // Set via JS or default

    // Resolve Category Slug to ID
    $catStmt = $pdo->prepare("SELECT id FROM categories WHERE slug = ?");
    $catStmt->execute([$categoryId]);
    $catRow = $catStmt->fetch();
    $realCategoryId = $catRow ? $catRow['id'] : null;

    if (!$title || !$realCategoryId) {
        $error = "Title and Category are required.";
    } else {
        try {
            $pdo->beginTransaction();

            // Handle Image Upload
            $imagePath = $article['featured_image'] ?? null;
            if (isset($_FILES['article-photo']) && $_FILES['article-photo']['error'] === UPLOAD_ERR_OK) {
                $uploadResult = uploadFile($_FILES['article-photo'], 'articles');
                if (isset($uploadResult['success'])) {
                    $imagePath = $uploadResult['filename'];
                } else {
                    throw new Exception($uploadResult['error']);
                }
            }

            $slug = generateSlug($title);
            // Append random string if slug exists and not editing same article
            // (Simplified slug logic for brevity)

            if ($articleId) {
                // UPDATE
                $sql = "UPDATE articles SET title=?, subheading=?, category_id=?, content=?, featured_image=?, image_caption=?, is_featured_top_story=?, status=?, meta_keywords=?, updated_at=NOW() WHERE id=?";
                $stmt = $pdo->prepare($sql);
                $stmt->execute([$title, $subheading, $realCategoryId, $content, $imagePath, $caption, $isTopStory, $status, $keywordsInput, $articleId]);
            } else {
                // INSERT
                $sql = "INSERT INTO articles (user_id, category_id, title, slug, subheading, content, featured_image, image_caption, status, is_featured_top_story, meta_keywords) VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?)";
                $stmt = $pdo->prepare($sql);
                $stmt->execute([$user['id'], $realCategoryId, $title, $slug, $subheading, $content, $imagePath, $caption, $status, $isTopStory, $keywordsInput]);
                $articleId = $pdo->lastInsertId();
            }

            // Handle Tags
            // 1. Clear existing tags for this article
            $pdo->prepare("DELETE FROM article_tags WHERE article_id = ?")->execute([$articleId]);
            
            // 2. Process new tags
            $tagsArray = array_map('trim', explode(',', $tagsInput));
            foreach ($tagsArray as $tagName) {
                if (empty($tagName)) continue;
                $tagSlug = generateSlug($tagName);
                
                // Check if tag exists
                $tCheck = $pdo->prepare("SELECT id FROM tags WHERE slug = ?");
                $tCheck->execute([$tagSlug]);
                $tagId = $tCheck->fetchColumn();

                if (!$tagId) {
                    $pdo->prepare("INSERT INTO tags (name, slug) VALUES (?, ?)")->execute([$tagName, $tagSlug]);
                    $tagId = $pdo->lastInsertId();
                }
                
                $pdo->prepare("INSERT INTO article_tags (article_id, tag_id) VALUES (?, ?)")->execute([$articleId, $tagId]);
            }

            $pdo->commit();
            
            if ($status === 'published') {
                // Update published_at if not set
                $pdo->prepare("UPDATE articles SET published_at = NOW() WHERE id = ? AND published_at IS NULL")->execute([$articleId]);
            }

            header("Location: preview.php?id=$articleId");
            exit;

        } catch (Exception $e) {
            $pdo->rollBack();
            $error = "Error saving article: " . $e->getMessage();
        }
    }
}

          // 2. Character Limit Logic
          function initCharLimit(inputId, counterId, errorId, limit) {
              const input = document.getElementById(inputId);
              const counter = document.getElementById(counterId);
              const errorMsg = document.getElementById(errorId);

              if (!input || !counter) return;

              function update() {
                  const len = input.value.length;
                  counter.textContent = `${len} / ${limit} chars`; // Explicitly set text

                  if (len >= limit) {
                      input.classList.add('limit-error');
                      errorMsg.classList.add('visible');
                  } else {
                      input.classList.remove('limit-error');
                      errorMsg.classList.remove('visible');
                  }
              }

              input.addEventListener('input', update);
              // Run once on load to populate count for existing data
              update();
          }

          initCharLimit('article-title', 'char-count-title', 'limit-msg-title', 100);
          initCharLimit('article-subheading', 'char-count-sub', 'limit-msg-sub', 200);
      });
