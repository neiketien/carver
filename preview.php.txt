<?php
require_once 'includes/config.php';
requireAuth();

$pdo = getDBConnection();
$user = getCurrentUser(); 

$pendingCount = 0;
if (hasRole('editor') || hasRole('admin')) {
    $pendingCount = $pdo->query("SELECT COUNT(*) FROM articles WHERE status = 'pending'")->fetchColumn();
}

$articleId = $_GET['id'] ?? null;

if (!$articleId) {
    header('Location: manage-articles.php');
    exit;
}

// Fetch Article Data
$stmt = $pdo->prepare("SELECT a.*, u.full_name as author_name FROM articles a JOIN users u ON a.user_id = u.id WHERE a.id = ?");
$stmt->execute([$articleId]);
$article = $stmt->fetch();

if (!$article) {
    die("Article not found.");
}

// Handle Publishing from Preview
if ($_SERVER['REQUEST_METHOD'] === 'POST' && isset($_POST['action'])) {
    $newStatus = $_POST['action'] === 'publish' ? 'published' : 'draft';
    
    // Only Editors/Admins can publish
    if ($newStatus === 'published' && !hasRole('editor')) {
        die("Unauthorized to publish.");
    }

    $updateSql = "UPDATE articles SET status = ?";
    if ($newStatus === 'published') {
        $updateSql .= ", published_at = IF(published_at IS NULL, NOW(), published_at)";
    }
    $updateSql .= " WHERE id = ?";
    
    $pdo->prepare($updateSql)->execute([$newStatus, $articleId]);
    
    header("Location: manage-articles.php");
    exit;
}
