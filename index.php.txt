<?php
require_once 'includes/config.php';

$pdo = getDBConnection();

// Fetch top story (is_featured_top_story = 1)
$stmtTopStory = $pdo->prepare("
    SELECT a.*, c.name as category_name, c.slug as category_slug, u.full_name as author_name,
           (SELECT t.slug FROM article_tags at JOIN tags t ON at.tag_id = t.id WHERE at.article_id = a.id LIMIT 1) as first_tag
    FROM articles a
    JOIN categories c ON a.category_id = c.id
    JOIN users u ON a.user_id = u.id
    WHERE a.status = 'published' AND a.is_featured_top_story = 1
    ORDER BY a.published_at DESC
    LIMIT 1
");

// Fetch top story (is_featured_top_story = 1)
//$stmtTopStory = $pdo->prepare("
    //SELECT a.*, c.name as category_name, c.slug as category_slug, u.full_name as author_name
    //FROM articles a
    //JOIN categories c ON a.category_id = c.id
    //JOIN users u ON a.user_id = u.id
    //WHERE a.status = 'published' AND a.is_featured_top_story = 1
    //ORDER BY a.published_at DESC
    //LIMIT 1
//");
$stmtTopStory->execute();
$topStory = $stmtTopStory->fetch();

// Fetch latest news (excluding top story)
$stmtLatest = $pdo->prepare("
    SELECT a.id, a.title, a.subheading, a.slug, a.featured_image, a.published_at,
           c.name as category_name, c.slug as category_slug,
           (SELECT t.slug FROM article_tags at JOIN tags t ON at.tag_id = t.id WHERE at.article_id = a.id LIMIT 1) as first_tag
    FROM articles a
    JOIN categories c ON a.category_id = c.id
    WHERE a.status = 'published' AND a.is_featured_top_story = 0
    ORDER BY a.published_at DESC
    LIMIT 6
");
$stmtLatest->execute();
$latestNews = $stmtLatest->fetchAll();

// Fetch most read articles (by view_count)
$stmtMostRead = $pdo->prepare("
    SELECT a.id, a.title, a.slug, a.featured_image, a.published_at, a.view_count,
           c.name as category_name, c.slug as category_slug,
           (SELECT t.slug FROM article_tags at JOIN tags t ON at.tag_id = t.id WHERE at.article_id = a.id LIMIT 1) as first_tag
    FROM articles a
    JOIN categories c ON a.category_id = c.id
    WHERE a.status = 'published'
    ORDER BY a.view_count DESC, a.published_at DESC
    LIMIT 8
");
$stmtMostRead->execute();
$mostRead = $stmtMostRead->fetchAll();

// Fetch trending articles by category
$trendingCategories = ['economy', 'tech', 'politics', 'energy'];
$trendingArticles = [];
foreach ($trendingCategories as $catSlug) {
    $stmt = $pdo->prepare("
        SELECT a.id, a.title, a.subheading, a.slug, a.featured_image, a.published_at,
               c.name as category_name, c.slug as category_slug,
               (SELECT t.slug FROM article_tags at JOIN tags t ON at.tag_id = t.id WHERE at.article_id = a.id LIMIT 1) as first_tag
        FROM articles a
        JOIN categories c ON a.category_id = c.id
        WHERE a.status = 'published' AND c.slug = ?
        ORDER BY a.published_at DESC
        LIMIT 5
    ");
    $stmt->execute([$catSlug]);
    $trendingArticles[$catSlug] = $stmt->fetchAll();
}

// Fetch opinion/analysis articles
$stmtOpinion = $pdo->prepare("
    SELECT a.id, a.title, a.slug, a.published_at,
           c.name as category_name, c.slug as category_slug,
           (SELECT t.slug FROM article_tags at JOIN tags t ON at.tag_id = t.id WHERE at.article_id = a.id LIMIT 1) as first_tag
    FROM articles a
    JOIN categories c ON a.category_id = c.id
    WHERE a.status = 'published' AND c.slug IN ('opinion', 'analysis')
    ORDER BY a.published_at DESC
    LIMIT 4
");
$stmtOpinion->execute();
$opinionArticles = $stmtOpinion->fetchAll();

// Helper function to build article URL
function buildArticleUrl($article) {
    $category = $article['category_slug'] ?? 'news';
    $tag = $article['first_tag'] ?? 'general';
    $slug = $article['slug'];
    return SITE_URL . "/$category/$tag/$slug";
}
