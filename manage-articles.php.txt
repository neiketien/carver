<?php
require_once 'includes/config.php';
requireAuth();

$pdo = getDBConnection();
$user = getCurrentUser();

// --- HANDLE ACTIONS (Trash, Restore, Permanent Delete) ---
if (isset($_GET['action']) && isset($_GET['id'])) {
    $actionId = $_GET['id'];
    $actionType = $_GET['action'];
    
    // Security check: Ensure user owns article or is admin/editor
    $checkSql = "SELECT user_id, status FROM articles WHERE id = ?";
    $checkStmt = $pdo->prepare($checkSql);
    $checkStmt->execute([$actionId]);
    $targetArticle = $checkStmt->fetch();

    if ($targetArticle) {
        $authorized = ($user['role'] === 'admin' || $user['role'] === 'editor' || $targetArticle['user_id'] == $user['id']);
        
        if ($authorized) {
            if ($actionType === 'trash') {
                // Soft Delete -> Move to Trash
                $pdo->prepare("UPDATE articles SET status = 'trash' WHERE id = ?")->execute([$actionId]);
                header("Location: manage-articles.php?success=Article moved to Paper Bin");
                exit;
            } elseif ($actionType === 'restore') {
                // Restore -> Move to Draft
                $pdo->prepare("UPDATE articles SET status = 'draft' WHERE id = ?")->execute([$actionId]);
                header("Location: manage-articles.php?success=Article restored to Drafts");
                exit;
            } elseif ($actionType === 'delete') {
                // Hard Delete -> Remove from DB
                $pdo->prepare("DELETE FROM articles WHERE id = ?")->execute([$actionId]);
                // Cleanup tags mapping
                $pdo->prepare("DELETE FROM article_tags WHERE article_id = ?")->execute([$actionId]);
                header("Location: manage-articles.php?success=Article permanently deleted");
                exit;
            }
        }
    }
}

// --- DATA FETCHING ---

$pendingCount = 0;
if (hasRole('editor') || hasRole('admin')) {
    $pendingCount = $pdo->query("SELECT COUNT(*) FROM articles WHERE status = 'pending'")->fetchColumn();
}

// Base Queries
$draftSql = "SELECT a.*, c.name as category_name FROM articles a JOIN categories c ON a.category_id = c.id WHERE a.status IN ('draft', 'pending', 'rejected')";
$pubSql = "SELECT a.*, c.name as category_name FROM articles a JOIN categories c ON a.category_id = c.id WHERE a.status = 'published'";
$trashSql = "SELECT a.*, c.name as category_name FROM articles a JOIN categories c ON a.category_id = c.id WHERE a.status = 'trash'";

$params = [];
if ($user['role'] === 'writer') {
    $filter = " AND a.user_id = ?";
    $draftSql .= $filter;
    $pubSql .= $filter;
    $trashSql .= $filter;
    $params[] = $user['id'];
}

$draftSql .= " ORDER BY a.updated_at DESC";
$pubSql .= " ORDER BY a.published_at DESC";
$trashSql .= " ORDER BY a.updated_at DESC";

// Execute
$stmt = $pdo->prepare($draftSql); $stmt->execute($params); $drafts = $stmt->fetchAll();

$stmt = $pdo->prepare($pubSql); $stmt->execute($params); $published = $stmt->fetchAll();

$stmt = $pdo->prepare($trashSql); $stmt->execute($params); $trash = $stmt->fetchAll();

<script>
    function switchTab(tab) {
        // 1. Hide all content sections
        document.querySelectorAll('.article-section').forEach(el => el.style.display = 'none');
        
        // 2. Show the targeted content
        document.getElementById(tab + 'Content').style.display = 'block';
        
        // 3. Reset ALL buttons to 'button-outline' (inactive)
        const buttons = document.querySelectorAll('.article-toggle-buttons button');
        buttons.forEach(btn => {
            btn.classList.remove('button-primary');
            btn.classList.add('button-outline');
        });

        // 4. Set the clicked button to 'button-primary' (active)
        const activeBtn = document.getElementById('btn-' + tab);
        if(activeBtn) {
            activeBtn.classList.remove('button-outline');
            activeBtn.classList.add('button-primary');
        }
    }
    
    document.addEventListener('DOMContentLoaded', () => {
        const alert = document.getElementById('successAlert');
        if (alert) {
            setTimeout(() => {
                alert.style.opacity = '0';
                setTimeout(() => alert.remove(), 500); // Wait for fade out
            }, 3000);
        }
    });

    function closeModals() {
        document.querySelectorAll('.modal-overlay').forEach(el => el.classList.remove('active'));
    }

            desc.innerText = "You are about to open the editor.";
            btn.innerText = "Open Editor";
        } else {
            title.innerText = "View Article";
            desc.innerText = "You are about to view this article.";
            btn.innerText = "View Page";
        }
        
        btn.href = url;
        modal.classList.add('active');
    }

    // Close on click outside
    window.onclick = function(event) {
        if (event.target.classList.contains('modal-overlay')) {
            closeModals();
        }
    }
