// Calculate KPIs
$publishedCount = $pdo->query("SELECT COUNT(*) FROM articles WHERE status = 'published'")->fetchColumn();
$pendingCount = $pdo->query("SELECT COUNT(*) FROM articles WHERE status = 'pending'")->fetchColumn();

if ($userRole === 'writer') {
    $draftCount = $pdo->prepare("SELECT COUNT(*) FROM articles WHERE status = 'draft' AND user_id = ?");
    $draftCount->execute([$_SESSION['user_id']]);
    $draftCount = $draftCount->fetchColumn();
} else {
    $draftCount = $pdo->query("SELECT COUNT(*) FROM articles WHERE status = 'draft'")->fetchColumn();
}

// Monthly views (last 30 days)
$monthlyViews = $pdo->query("
    SELECT SUM(view_count) 
    FROM articles 
    WHERE status = 'published' 
    AND published_at >= DATE_SUB(NOW(), INTERVAL 30 DAY)
")->fetchColumn() ?? 0;

// Recent activity based on role
if ($userRole === 'writer') {
    $recentArticles = $pdo->prepare("
        SELECT a.*, c.name as category_name, a.status as article_status
        FROM articles a
        JOIN categories c ON a.category_id = c.id
        WHERE a.user_id = ?
        ORDER BY a.updated_at DESC
        LIMIT 4
    ");
    $recentArticles->execute([$_SESSION['user_id']]);
} else {
    $recentArticles = $pdo->query("
        SELECT a.*, c.name as category_name, u.full_name as author_name, a.status as article_status
        FROM articles a
        JOIN categories c ON a.category_id = c.id
        JOIN users u ON a.user_id = u.id
        ORDER BY a.updated_at DESC
        LIMIT 4
    ");
}
$recentArticles = $recentArticles->fetchAll();

// Notifications (pending articles for editors/admins)
$notifications = [];
if (hasRole('editor')) {
    $notifStmt = $pdo->prepare("
        SELECT n.*, a.title as article_title, a.slug as article_slug
        FROM notifications n
        LEFT JOIN articles a ON n.article_id = a.id
        WHERE n.user_id = ? AND n.is_read = 0
        ORDER BY n.created_at DESC
        LIMIT 5
    ");
    $notifStmt->execute([$_SESSION['user_id']]);
    $notifications = $notifStmt->fetchAll();
}

// Team activity (for admins/editors)
if (hasRole('editor')) {
    $teamActivity = $pdo->query("
        SELECT u.full_name, u.email, u.role,
               (SELECT title FROM articles WHERE user_id = u.id AND status = 'published' ORDER BY published_at DESC LIMIT 1) as recent_article,
               (SELECT updated_at FROM articles WHERE user_id = u.id ORDER BY updated_at DESC LIMIT 1) as last_activity
        FROM users u
        WHERE u.status = 'active'
        ORDER BY last_activity DESC
        LIMIT 5
    ");
    $teamActivity = $teamActivity->fetchAll();
}

// Users list (admins only)
if (hasRole('admin')) {

        ORDER BY created_at DESC
