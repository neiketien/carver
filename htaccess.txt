RewriteEngine On
RewriteBase /

# ----------------------------------------------------------------------
# 1. API WHITELIST (CRITICAL FIX)
# ----------------------------------------------------------------------
# Stop processing redirects for these specific files
RewriteRule ^api/newsletter\.php$ - [L]
RewriteRule ^api/load-more\.php$ - [L]

# ----------------------------------------------------------------------
# 2. FORCE HTTPS
# ----------------------------------------------------------------------
RewriteCond %{HTTPS} off
RewriteRule ^(.*)$ https://%{HTTP_HOST}%{REQUEST_URI} [L,R=301]

# ----------------------------------------------------------------------
# 3. HSTS
# ----------------------------------------------------------------------
<IfModule mod_headers.c>
    Header always set Strict-Transport-Security "max-age=31536000; includeSubDomains; preload"
</IfModule>

# Prevent directory browsing
Options -Indexes

# Protect files
<FilesMatch "^\.">
    Order allow,deny
    Deny from all
</FilesMatch>
<FilesMatch "(config|\.env)\.php$">
    Order allow,deny
    Deny from all
</FilesMatch>

# ----------------------------------------------------------------------
# 4. REMOVE .php EXTENSION
# ----------------------------------------------------------------------
# Only redirect if it is NOT an API file (Safety check)
RewriteCond %{REQUEST_URI} !^/api/
RewriteCond %{THE_REQUEST} \s/+(.+?)\.php[\s?] [NC]
RewriteRule ^ /%1 [R=301,L,NE]

# ----------------------------------------------------------------------
# 5. HANDLE EXTENSIONLESS URLS
# ----------------------------------------------------------------------
RewriteCond %{REQUEST_FILENAME} !-d
RewriteCond %{REQUEST_FILENAME}.php -f
RewriteRule ^([^\.]+)$ $1.php [NC,L]

# ----------------------------------------------------------------------
# 6. CUSTOM ROUTES
# ----------------------------------------------------------------------
RewriteRule ^(politics|economy|tech|energy|opinion|analysis)/?$ category.php?slug=$1 [L,QSA]
RewriteRule ^latest-news/?$ latest-news.php [L,QSA]
RewriteRule ^([a-z-]+)/([a-z0-9-]+)/([a-z0-9-]+)/?$ article.php?category=$1&tag=$2&slug=$3 [L,QSA]
RewriteRule ^admin-([a-z-]+)\.php$ admin-$1.php [L]

# Custom Error Pages
ErrorDocument 404 /404.php
ErrorDocument 500 /404.php

# ----------------------------------------------------------------------
# 7. TAG ROUTING
# ----------------------------------------------------------------------
RewriteRule ^tag/([a-z0-9-]+)/?$ tag.php?slug=$1 [L,QSA]
